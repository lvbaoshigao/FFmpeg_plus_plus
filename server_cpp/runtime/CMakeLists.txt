# WAMR is compiled into ffmpegpp; no runtime download or extra shared library.
include(FetchContent)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(FFMPEGPP_WAMR_SOURCE_DIR "" CACHE PATH "Offline WAMR 2.4.6 source override")
if(FFMPEGPP_WAMR_SOURCE_DIR)
  set(FETCHCONTENT_SOURCE_DIR_FFMPEGPP_WAMR "${FFMPEGPP_WAMR_SOURCE_DIR}")
endif()
FetchContent_Declare(ffmpegpp_wamr
  URL https://codeload.github.com/wasm-micro-runtime/wasm-micro-runtime/tar.gz/0fa421193212b3fb45c7f59ec53662cc2851fcc7
  URL_HASH SHA256=b88e781ec4b27fc568de0f758a87dd1be13e8b275efaa32bcf9d5046493dc9e7
  # Download only: upstream root CMake configures an unrelated CLI product.
  SOURCE_SUBDIR ffmpegpp-embed-source-only
)
FetchContent_MakeAvailable(ffmpegpp_wamr)
set(WAMR_ROOT_DIR "${ffmpegpp_wamr_SOURCE_DIR}")
if(ANDROID)
  set(WAMR_BUILD_PLATFORM android)
elseif(WIN32)
  set(WAMR_BUILD_PLATFORM windows)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  set(WAMR_BUILD_PLATFORM darwin)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(WAMR_BUILD_PLATFORM linux)
else()
  message(FATAL_ERROR "WAMR plugins have not been validated for ${CMAKE_SYSTEM_NAME}; configure -DFFMPEGPP_ENABLE_PLUGINS=OFF")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64|ARM64)")
  set(WAMR_BUILD_TARGET AARCH64)
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(WAMR_BUILD_TARGET X86_64)
else()
  message(FATAL_ERROR "Plugins currently support x86_64 and arm64 hosts")
endif()
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_INSTRUCTION_METERING 1)
# Async terminate needs the thread manager; guest thread APIs remain absent.
set(WAMR_BUILD_THREAD_MGR 1)
foreach(feature AOT JIT FAST_JIT LIBC_BUILTIN LIBC_WASI LIBC_UVWASI LIB_PTHREAD LIB_WASI_THREADS SHARED_MEMORY MULTI_MODULE MULTI_MEMORY MEMORY64 SIMD REF_TYPES GC STRINGREF TAIL_CALL DEBUG_INTERP MINI_LOADER)
  set(WAMR_BUILD_${feature} 0)
endforeach()
set(WAMR_BUILD_BULK_MEMORY 1)
# Software checks avoid installing guest-memory fault handlers in Flutter.
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
include("${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake")
add_library(ffmpegpp_wamr_runtime STATIC ${WAMR_RUNTIME_LIB_SOURCE})
set_target_properties(ffmpegpp_wamr_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(ffmpegpp_wamr_runtime PUBLIC "${WAMR_ROOT_DIR}/core/iwasm/include")
if(WIN32)
  target_link_libraries(ffmpegpp_wamr_runtime PUBLIC ntdll)
  if(MINGW)
    target_link_libraries(ffmpegpp_wamr_runtime PUBLIC ws2_32)
  endif()
elseif(ANDROID)
  target_link_libraries(ffmpegpp_wamr_runtime PUBLIC m ${CMAKE_DL_LIBS} android log)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(ffmpegpp_wamr_runtime PUBLIC m ${CMAKE_DL_LIBS} Threads::Threads)
endif()
